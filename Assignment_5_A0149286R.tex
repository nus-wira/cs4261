\subsection{}
From $S$, we have
\[0 \leq y^2 \leq 1-\frac{x^2}{2}\]
\[0 \leq x \leq \sqrt{2}\]
\[0 \leq y \leq \sqrt{1-\frac{x^2}{2}}\]
\subsubsection{Socially Optimal}
We want to maximize $x+y$.
\[x + y \leq x + \sqrt{1-\frac{x^2}{2}}\]
Let $f(x)=x + \sqrt{1-\frac{x^2}{2}}$.
\[f'(x)=1-\frac{x}{2\sqrt{1-\frac{x^2}{2}}}\]
\[f''(x)=-\frac{\sqrt{2}}{(2-x^2)^{3/2}}<0\]
We have a maximum when $f'(x)=0$.
\[1-\frac{x}{2\sqrt{1-\frac{x^2}{2}}}=0\]
\[4-2x^2=x^2\]
\[x=\frac{2}{\sqrt{3}}\]
Then we have
\[y=\sqrt{1-\frac{x^2}{2}}=\frac{1}{\sqrt{3}}\]
\[x+y=\frac{3}{\sqrt{3}}=\sqrt{3}\]

\subsubsection{Nash Bargaining}
We want to maximize $xy$.
\[xy \leq x\sqrt{1-\frac{x^2}{2}}\]
Let $f(x)=x \sqrt{1-\frac{x^2}{2}}$.
\[f'(x)=-\frac{x^2-1}{\sqrt{1-\frac{x^2}{2}}}\]
\[f''(x)=\frac{\sqrt{2}x(x^2-3)}{(2-x^2)^{3/2}}<0\]
We have a maximum when $f'(x)=0$.
\[-\frac{x^2-1}{\sqrt{1-\frac{x^2}{2}}}=0\]
\[x=1\]
Then
\[y=\sqrt{1-\frac{x^2}{2}}=\frac{1}{\sqrt{2}}\]
\[xy=\frac{1}{\sqrt{2}}\]

\section{Network Flow Games}

\subsection{Superadditive}
Suppose we have disjoint sets $S,T\subseteq N$.
Since any flow using edges in $S$ will be in $S\cup T$,
and any flow using edges in $T$ will be in $S\cup T$,
and all flows using edges only in $S$ will not be a flow
using edges only in $T$, and vice versa, 
it can be seen that
\[v(S) + v(T) \leq v(S\cup T)\]
The game is superadditive.


\subsection{Core}

\begin{theorem}
    Max-Flow Min-Cut.
    The maximum amount of flow from $s$ to $t$ is equal to
    the total weight of edges in a minimum cut.
\end{theorem}
\begin{corollary}
    The maximum amount of flow from $s$ to $t$ is less than or equal to
    the total weight of edges in any cut.
\end{corollary}
Consider a minimum cut $C\subseteq N$.
From the theorem, and setting $x_i=w_i,\forall i\in C$, 
and $x_i=0$ otherwise, we have
\[\sum_{i\in N}x_i=\sum_{i\in C}w_i = v(N)\]
Consider any subset $S\subseteq N$.
Then we know since $C$ is a minimum cut of the larger graph $N$,
$S\cap C$ is a cut for the smaller graph $S$.
\[\sum_{i\in S}x_i = \sum_{i\in S\cap C}w_i \geq v(S)\]
Thus, with these two conditions, we have a valid core.
The core of the network flow game is non-empty.

\section{Coalition Structure}

\subsection{Lean}
Suppose we have an optimal coalition structure $CS^*$ that is not lean.
Clearly an optimal coalition structure exists,
since players can always be partitioned in a weighted voting game.
We can take two steps.
\begin{enumerate}
    \item If it has more than one losing coalition, 
    simply union the losing coalitions 
    to create a larger losing coalition $L\in CS^*$.
    Since $CS^*$ is optimal, this will not create any new winning coalition.
    \item Then, $\forall S\in CS_+^*$, 
    if $\exists i\in S, S\backslash\{i\}$ is winning,
    move $i$ to the losing coalition $L$. 
    With the same reasoning as Step 1, 
    this will not create any new winning coalition.
    Continue to remove players until each $S\in CS_+^*$
    satisfies the lean condition.
\end{enumerate}
Thus, we have a resulting lean optimal coalition structure.

\subsection{}
Suppose not, that $CS^*$ is lean, optimal but
\[\exists S\in CS_+^*, w(S)\geq 2q \wedge |S| \geq 2\]
Since $CS^*$ is lean, removing any player $i\in S$ will result
in a losing coalition $S\backslash\{i\}$.
That is,
\[\forall i\in S:w(S\backslash\{i\})=w(S)-w_i<q\]
\[\forall i\in S:w_i>w(S)-q\geq 2q-q=q\]
But this is a contradiction, since any player $i\in S$ alone 
results in a winning coalition, making $CS^*$ not lean.
Thus, the supposition is false.
If $CS^*$ is a lean, optimal coalition structure for a WVG, then
\[\forall S\in CS_+^*: |S| \geq 2 \rightarrow w(S) < 2q\]

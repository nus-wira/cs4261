\section{Two Firms, Same Good}

\subsection{Strategy Profile}
Let $f(q_1,q_2)=u_1(q_1,q_2)+u_2(q_1,q_2)$. Assume $a\geq q_1+q_2$.
\begin{equation}
\begin{aligned}
    f(q_1,q_2)
    &= q_1\cdot p(q_1,q_2) - cq_1 + q_2\cdot p(q_1,q_2) - cq_2\\
    &= (q_1+q_2)(p(q_1,q_2)-c)\\
    &= (q_1+q_2)(a-(q_1+q_2)-c)
\end{aligned}
\end{equation}
\begin{equation}
\begin{aligned}
    \frac{\partial f}{\partial q_1} &= \frac{\partial f}{\partial q_2} 
    \quad\text{(by symmetry)}\\
    &= (a-(q_1+q_2)-c)-(q_1+q_2)\\
    &= a-2(q_1+q_2)-c\\
\end{aligned}
\end{equation}
At the critical point,
\[\frac{\partial f}{\partial q_1}= \frac{\partial f}{\partial q_2} =0\]
Thus,
\begin{equation}
    q_1+q_2=\frac{a-c}{2}
\end{equation}
Since
\begin{equation}
\begin{aligned}
    \frac{\partial^2 f}{\partial^2 q_1} 
    = \frac{\partial^2 f}{\partial^2 q_2} 
    = \frac{\partial^2 f}{\partial q_1\partial q_2} 
    = -2 < 0
\end{aligned}
\end{equation}
Hence, the line in $(3)$ is the maximum.\\
It can be shown that the assumption $a\geq q_1+q_2$ will always be true.
\[a \geq \frac{a-c}{2}\]
\[\frac{a}{2} \geq -\frac{c}{2}\]
which is always true as $a>c>0$.
Thus, the strategy profile
\[(q_1,q_2) 
=s\cdot\left(\frac{a-c}{2},0\right)+(1-s)\cdot\left(0,\frac{a-c}{2}\right),
s\in[0,1]\]
will maximize the joint profit.

\subsection{Nash Equilibrium}
Assume that $q_1+q_2\leq a$.
Since 
\[BR_1(q_2) = \max_{q_1\in\Delta_1}u_1(q_1,q_2)\]
it suffices to find where $\frac{\partial u_1}{\partial q_1} = 0$,
with $\frac{\partial^2 u_1}{\partial q_1^2} = -2 < 0$.
\[u_1(q_1,q_2) = q_1(a-(q_1+q_2)-c)\]
\begin{equation}
\begin{aligned}
    \frac{\partial u_1}{\partial q_1} &= (a-(q_1+q_2)-c) - q_1\\
    &= -2q_1 + a-q_2-c
\end{aligned}
\end{equation}
Thus,
\[-2q_1 + a-q_2-c=0\]
\[q_1=\frac{a-c-q_2}{2}\]
Then
\[BR_1(q_2)=\frac{a-c-q_2}{2}\]
Similarly,
\[BR_2(q_1)=\frac{a-c-q_1}{2}\]
Then in the Nash equilibrium, we have
\begin{align}
    q_1=\frac{a-c-q_2}{2}\\
    q_2=\frac{a-c-q_1}{2}
\end{align}
Solving,
\[q_1=\frac{a-c-\frac{a-c-q_1}{2}}{2}\]
\[q_1=\frac{a-c+q_1}{4}\]
\[q_1=\frac{a-c}{3}\]
By symmetry,
\[q_1=q_2=\frac{a-c}{3}\]
It can be shown that the assumption $a\geq q_1+q_2$ will always be true.
\[a \geq \frac{2(a-c)}{3}\]
\[\frac{a}{3} \geq -\frac{c}{3}\]
which is always true as $a>c>0$.
The Nash equilibrium is
\[(q_1,q_2)=\left(\frac{a-c}{3},\frac{a-c}{3}\right)\]

\section{Proving Nash Equilibria}
\subsection{Dominate}
True.\\
\begin{proof}
Suppose not, that action $a$ dominates action $b$ for player $i$,
and action $b$ is played with probability $p_b>0$ in some Nash Equilbrium.
Since $a$ strictly dominates $b$,
\begin{equation}
    \forall \vec{p}_{-i}\in\Delta(A_{-i}): 
    u_i(\vec{p}_{-i},a)> u_i(\vec{p}_{-i},b)
\end{equation}
By definition of Nash equilibrium,
\begin{equation}
    \forall i\in N,\forall \vec{q}_i\in\Delta(A_i):
    u_i(\vec{p})\geq u_i(\vec{p}_{-i},\vec{q}_i)
\end{equation}
Also, $\exists \vec{p}_i^*\in\Delta(A_i)$ 
such that $p_b^* > 0$ for $\vec{p}_i^*$, which is part of a Nash equilibrium.
Let the probability of action $a$ being played in $\vec{p}_i^*$ be $p_a^*$.\\
Let $\vec{p}^*$ be the Nash equlibrium strategy profile for which 
$\vec{p}_i^*$ is the strategy profile for player $i$.\\
Let $\vec{p}^{**}$ then be the strategy profile where we set
$p_a^{**}=p_a^*+p_b^*$ and $p_b^{**}=0$ in $\vec{p}_i^{**}$, ceteris paribus.\\
From $(8)$, we have that
\begin{equation}
    (p_a^*+p_b^*)u_i(\vec{p}_{-i},a) 
    > p_a^* u_i(\vec{p}_{-i},a) + p_b^* (\vec{p}_{-i},b)
\end{equation}
Then,
\begin{equation}
    u_i(\vec{p}^{**}) > u_i(\vec{p}^*)
\end{equation}
Combining $(9),(11)$, with $\vec{p}^*$ being a Nash equilibrium,
\begin{equation}
    u_i(\vec{p}^{**}) > u_i(\vec{p}^*) \geq u_i(\vec{p}^{**})
\end{equation}
A contradiction, so $b$ must be played with probability 0 
in any Nash equilibrium.
\end{proof}

\subsection{Best Response, Mixed Strategy}
True.\\
\begin{proof}
The best response is
\[BR_i(\vec{p}_{-i})=\argmax_{\vec{q}\in\Delta_i}u_i(\vec{p}_{-i},\vec{q})\]
\begin{equation}
    \forall c\in BR_i(\vec{p}_{-i}),\forall\vec{q}\in\Delta_i: 
    u_i(\vec{p}_{-i},c)\geq u_i(\vec{p}_{-i},\vec{q})
\end{equation}
Since $a,b\in A_i$ are both best responses,
\begin{equation}
    \max_{\vec{q}\in\Delta_i}u_i(\vec{p}_{-i},\vec{q})
    =u_i(\vec{p}_{-i},a)=u_i(\vec{p}_{-i},b)
\end{equation}
Let $\vec{p}_i^*$ be a strategy by player $i$ where
actions $a,b$ are played with $\gamma,1-\gamma$ probability 
respectively, $0<\gamma < 1$. 
For $\vec{p}_i^*$ to be a best response, it suffices to show that 
\[u_i(\vec{p}_{-i},\vec{p}_i^*) 
= \max_{\vec{q}\in\Delta_i}u_i(\vec{p}_{-i},\vec{q})\]
\begin{equation}
\begin{aligned}
    u_i(\vec{p}_{-i},\vec{p}_i^*) 
    &= \gamma u_i(\vec{p}_{-i},a) + (1-\gamma) u_i(\vec{p}_{-i},b)\\
    &= \gamma u_i(\vec{p}_{-i},a) + (1-\gamma) u_i(\vec{p}_{-i},a)
    \quad\text{(from (14))}\\
    &= u_i(\vec{p}_{-i},a)\\
    &= \max_{\vec{q}\in\Delta_i}u_i(\vec{p}_{-i},\vec{q})
    \quad\text{(from (14))}
\end{aligned}
\end{equation}
Thus, we have $\vec{p}_i^*$ as a best response.
\end{proof}

\subsection{Brouwer's Fixed Point Theorem}

Consider the function $g(x)=f(x)-x$.
Since $f(x)$ is continuous, so is $g(x)$.
\begin{claim}
For any continuous function $f:[0,1]\rightarrow [0,1]$,
    \[\exists x_0\in [0,1], g(x_0) = f(x_0)-x = 0\]
\end{claim}
\begin{proof}
    If $f(0)=0$ or $f(1)=1$, then $g(0)=0$ or $g(1)=1$.
    Otherwise, we have $f(0)\neq 0$ and $f(1) \neq 1$.
    Since $f(x)\in [0,1]$, then $f(0) > 0,f(1) < 1$, so
    \[g(0) = f(0)-0>0\]
    \[g(1) = f(1)-1<0\]
    \[g(1)<0<g(0)\]
    Then, by the Intermediate Value Theorem,
    \[\exists x_0 \in (0,1), g(x_0) = 0\]
    In all cases,
    \[\exists x_0 \in [0,1], g(x_0) = 0\]
\end{proof}\\
Thus,
\[\exists x_0 \in [0,1], f(x_0)=x_0\]

\subsection{Non-Empty, Convex, and Compact Domain}

\subsubsection{Non-Empty}
For all action spaces $\Delta(A_i)$, choose $\vec{p}_i\in\Delta(A_i)$.
$p_i$ exists, since the action spaces are not empty.
Thus, $(\vec{p}_1,\vec{p}_2,\dots,\vec{p}_n)\in\Delta(A)$ and $\Delta(A)$ is non-empty.

\subsubsection{Convex}
Let $\vec{a},\vec{b}\in \Delta(A), 
\vec{a}=(\vec{a}_1,...,\vec{a}_n),\vec{b}=(\vec{b}_1,...\vec{b}_n)$
Define 
\[\vec{c}=t\vec{a}+(1-t)\vec{b}
=(t\vec{a}_1+(1-t)\vec{b}_1,...,t\vec{a}_n+(1-t)\vec{b}_n)\]
$\Delta(A)$ is convex if,
\[\forall \vec{a},\vec{b}\in \Delta(A),
\forall t\in[0,1]:\vec{c}=t\vec{a}+(1-t)\vec{b}\in\Delta(A)\]
It suffices to show that $\vec{c}_i\in\Delta(A_i)$
for an arbitrary player $i$.
We have $\vec{c}_i\in\Delta(A_i)$ if, 
given player $i$ has $m$ possible actions,
\[\vec{c}_i\in[0,1]^m \wedge sum\{\vec{c}_i\}=1\]
For any action $j$ with probability 
$p_{aj},p_{bj}\in[0,1]$ in $\vec{a}_i,\vec{b}_i$ respectively,
\[t\cdot p_{aj} + (1-t) \cdot p_{bj} \in [0,1]\]
So,
\[\vec{c}_i\in[0,1]^m\]
Now,
\begin{equation}
\begin{aligned}
    sum\{\vec{c}_i\} &= sum\{t\vec{a}_i+(1-t)\vec{b}_i\}\\
    &= t\cdot sum\{\vec{a}_i\} + (1-t)\cdot sum\{\vec{b}_i\}\\
    &= t\cdot 1 + (1-t) \cdot 1\\
    &= 1
\end{aligned}
\end{equation}
Thus, we have $\vec{c}_i\in\Delta(A_i)$ for an arbitrary player $i$,
and thus, $\Delta(A)$ is convex.

\subsubsection{Compact}
Take any arbitrary probability distribution over player $i$'s actions,
$\Delta(A_i)$. Since it is a probability space over a finite domain,
then $\Delta(A_i)$ is compact. Thus, $\forall i\in N,\Delta(A_i)$ is compact.\\
We then prove by induction that 
$\Delta(A)=\Delta(A_1)\times\dots\times\Delta(A_n)$ is compact.\\
\begin{proof}
    Base Case $(n=1)$. $\Delta(A_1)$ is compact.\\
    Induction Case. Suppose $\Delta(A_1)\times\dots\times\Delta(A_k)$ is compact.
    Then 
    \[\Delta(A_1)\times\dots\times\Delta(A_k)\times\Delta(A_{k+1})\]
    is also compact.
\end{proof}\\
Thus, $\Delta(A)$ is compact.\\
It is shown that $\Delta(A)$ is non-empty, convex, and compact.

\subsection{Mixed Strategy}

To prove that $s_i(\vec{p}) = (s_i(\vec{p},a))_{a\in A_i}$
is a mixed strategy over $A_i$, it suffices to show that,
\[(\forall a\in A_i : s_i(\vec{p},a)\in[0,1]) \wedge 
\sum_{a\in A_i} s_i(\vec{p},a)=1\]
We first prove $\forall a\in A_i : s_i(\vec{p},a)\in[0,1]$.
It can be easily seen that $G_i(\vec{p},a) \geq 0$,
and thus $s_i(\vec{p},a)\geq 0$. We also have
\[p_i(a)+G_i(\vec{p},a) \leq 1 +G_i(\vec{p},a) 
\leq 1 +\sum_{a\in A_i} G_i(\vec{p},a)\]
Thus, $s_i(\vec{p},a)\leq 1$. 
\[\forall a\in A_i : s_i(\vec{p},a)\in[0,1]\]
\begin{equation}
\begin{aligned}
    \sum_{a\in A_i} s_i(\vec{p},a) 
    &= \sum_{a\in A_i} 
    \frac{p_i(a)+G_i(\vec{p},a)}{1 +\sum_{a\in A_i} G_i(\vec{p},a)}\\
    &= \frac{\sum_{a\in A_i} \left(p_i(a)+G_i(\vec{p},a)\right)}
    {1 +\sum_{a\in A_i} G_i(\vec{p},a)}\\
    &= \frac{\sum_{a\in A_i} p_i(a)+\sum_{a\in A_i}G_i(\vec{p},a)}
    {1 +\sum_{a\in A_i} G_i(\vec{p},a)}\\
    &= \frac{1 +\sum_{a\in A_i} G_i(\vec{p},a)}
    {1 +\sum_{a\in A_i} G_i(\vec{p},a)}\\
    &= 1
\end{aligned}
\end{equation}
Thus, $s_i(\vec{p}) = (s_i(\vec{p},a))_{a\in A_i}$
is a mixed strategy over $A_i$. 
Then, 
\[\forall i\in N:s_i(\vec{p})\in\Delta(A_i)\]
Then $(s_1(\vec{p}),\dots,s_n(\vec{p}))\in\Delta(A)$.
Then we have 
\[\beta : \Delta(A) \rightarrow \Delta(A)\] 
i.e. the function $\beta$ maps from $\Delta(A)$ to itself.


\subsection{Continuous}
It suffices to show that $s_i(\vec{p},a)$ is continuous, as this results in
$s_i(\vec{p})$ and subsequently $\beta$ being continuous as well.
\begin{claim}
    $s_i(\vec{p},a)$ is continuous
\end{claim}
\begin{proof}
    Given that $\vec{p}\in\Delta(A)$, 
    which is a non-empty, convex, and compact domain,
    then $p_i(a)$ is continuous.
    $u_i(\vec{p}_{-i},a),u_i(\vec{p})$ are also continuous, 
    which are utility functions with
    \[u_i(\vec{p})=\sum_{\vec{a}\in A}u_i(\vec{a})\Pr[\vec{a}]\]
    where $u_i(\vec{a})$ is a constant that is trivially continuous.
    Then,
    \[G_i(\vec{p},a)=\max\{u_i(\vec{p}_{-i},a)-u_i(\vec{p}),0\}\]
    is continuous, and so
    \[p_i(a)+G_i(\vec{p},a)\]
    \[1+\sum_{a\in A_i}G_i(\vec{p},a)\]
    are continuous as well. 
    Then $s_i(\vec{p},a)$ is continuous.
\end{proof}\\
Thus, $\beta$ is continuous.

\subsection{Nash Equilibrium}

We have the non-empty, convex, compact domain $D=\Delta(A)$,
and the continuous function $f=\beta$.
By Brouwer's Fixed Point Thoerem,
\begin{corollary}
    \[\exists \vec{p}^*\in\Delta(A), \beta(\vec{p}^*)=\vec{p}^*\]
\end{corollary}
\begin{lemma}
    \[\forall i\in N,\forall \vec{p}\in\Delta(A) : 
    u_i(\vec{p}_{-i},s_i(\vec{p})) \geq u_i(\vec{p})\]
\end{lemma}
\begin{proof}
    Notice that $\forall a\in A_i$,
    \[u_i(\vec{p}_{-i},a)\geq u_i(p) \Leftrightarrow 
    s_i(\vec{p},a) \geq p_i(a)\]
    Then 
    \begin{equation}
    \begin{aligned}
    u_i(\vec{p}_{-i},a) 
    &= \sum_{a\in A_i} s_i(\vec{p},a)\cdot u_i(\vec{p}_{-i},a)\\
    &\geq \sum_{a\in A_i} p_i(a)\cdot u_i(\vec{p}_{-i},a)\\
    &= u_i(\vec{p})
    \end{aligned}
    \end{equation}
\end{proof}\\
By \textbf{Corollary 3} we have for some $\vec{p}^*\in\Delta(A)$,
\[\forall i\in N : \beta(\vec{p}^*)_{-i}=\vec{p}_{-i}^*\]
Then we can set $\beta(\vec{p}^*)=\vec{p}^*
=(\vec{p}_{-i},s_i(\vec{p}))$, and by \textbf{Lemma 4},
\[\forall i\in N, \forall \vec{q}_i\in\Delta(A_i) :
u_i(\vec{p}^*)\geq u_i(\vec{p}_{-i}^*,\vec{q}_i)\]
In other words, $\vec{p}^*$ is a Nash equilibrium.

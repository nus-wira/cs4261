\section{Two Firms, Same Good}

\subsection{Strategy Profile}
Let $f(q_1,q_2)=u_1(q_1,q_2)+u_2(q_1,q_2)$. Assume $a>q_1+q_2$.
\begin{equation}
\begin{aligned}
    f(q_1,q_2)
    &= q_1\cdot p(q_1,q_2) - cq_1 + q_2\cdot p(q_1,q_2) - cq_2\\
    &= (q_1+q_2)(p(q_1,q_2)-c)\\
    &= (q_1+q_2)(a-(q_1+q_2)-c)
\end{aligned}
\end{equation}
\begin{equation}
\begin{aligned}
    \frac{\partial f}{\partial q_1} &= \frac{\partial f}{\partial q_2} 
    \quad\text{(by symmetry)}\\
    &= (a-(q_1+q_2)-c)-(q_1+q_2)\\
    &= a-2(q_1+q_2)-c\\
\end{aligned}
\end{equation}
At the critical point,
\[\frac{\partial f}{\partial q_1}= \frac{\partial f}{\partial q_2} =0\]
Thus,
\begin{equation}
    q_1+q_2=\frac{a-c}{2}
\end{equation}
Since
\begin{equation}
\begin{aligned}
    \frac{\partial^2 f}{\partial^2 q_1} 
    = \frac{\partial^2 f}{\partial^2 q_2} 
    = \frac{\partial^2 f}{\partial q_1\partial q_2} 
    = -2 < 0
\end{aligned}
\end{equation}
Hence, the line in $(3)$ is the maximum.\\
If $a\leq p_1+p_2$, then 
\[f(q_1,q_2)=-c(q_1+q_2)\leq 0\]
Trivially, the strategy profile
\[(q_1,q_2)=(0,0)\]
maximizes the joint profit (none).\\
Thus, the strategy profile
\[(q_1,q_2) =
    \begin{cases}
        s\cdot(1,-1)+(0,\frac{a-c}{2}),s\in\mathbb{R}^+ 
        & \text{if } a>q_1+q_2\\
        (0,0), & \text{otherwise}
    \end{cases}
\]
will maximize the joint profit.

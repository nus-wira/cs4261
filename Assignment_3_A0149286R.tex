\section{Regret, Small Number of Rounds}
We want the expected loss at each time step to be at least 1/2,
so that the total loss is at least $T/2$.
We assign the losses as follows.
\begin{enumerate}
    \item At $t=1$, we start by choosing $n/2$ experts to have 1 loss
    based on whatever the algorithm $\mathcal{A}$ picked.
    \item We maintain that half the experts from time step $t$
    that had 0 loss will have 1 loss in time step $t+1$,
    while those that had 1 loss at $t$ will continue to have 1 loss at $t+1$.
    These assignments are similarly based on the algorithm $\mathcal{A}$
    to ensure maximal loss.
    We can see that even if the algorithm tries to focus on experts that
    never had loss up to time step $t$, half of them will end up with
    loss at $t+1$.
    The rest of the experts have loss 1 anyway.
\end{enumerate}
Next, we notice that the number of experts with 0 loss halves at 
every time step.
This means, at the end of $T$ time steps, there are
\[\frac{n}{2^T}>\frac{n}{n}=1\]
experts that have maintained 0 loss for all $T$ time steps.
This satisfies (b), and (a) is already satisfied by having
at least 1/2 expected loss for $T$ time steps.

\section{Routing Game}

\subsection{Optimal Flow}
The social cost $c$ is
\[c=\varepsilon+(1-\varepsilon)^{p+1}\]
Then
\begin{equation}
\begin{aligned}
    \frac{dc}{d\varepsilon}=1-(p+1)(1-\varepsilon)^p\\
    \frac{d^2c}{d\varepsilon^2}=p(p+1)(1-\varepsilon)^{p-1} > 0
\end{aligned}
\end{equation}
We have a minimum when $\frac{dc}{d\varepsilon}=0$.
\[1-(p+1)(1-\varepsilon_0)^p=0\]
\[(1-\varepsilon_0)^p=\frac{1}{p+1}\]
\begin{equation}
    \varepsilon_0=1-\frac{1}{\sqrt[p]{p+1}}
\end{equation}
We also have
\[PoA(\mathcal{G}_p)=\frac{WorstNash(\mathcal{G}_p)}{OPT(\mathcal{G}_p)}\]
where
\[WorstNash(\mathcal{G}_p)=1\]
\begin{equation}
\begin{aligned}
    OPT(\mathcal{G}_p)&=c(\varepsilon=\varepsilon_0)\\
    &=1-(p+1)^{-\frac{1}{p}}+(p+1)^{-\frac{p+1}{p}}\\
    &=1+(p+1)^{-\frac{1}{p}}\left[(p+1)^{-1}-1\right]
\end{aligned}
\end{equation}
Then
\begin{equation}
    PoA(\mathcal{G}_p)
    =\frac{1}{1+(p+1)^{-\frac{1}{p}}\left[(p+1)^{-1}-1\right]}
\end{equation}
\subsection{Ratio Limit}
Note that
\begin{equation}
\begin{aligned}
    \lim_{p\rightarrow\infty}(p+1)^{-\frac{1}{p}}
    &=\lim_{p\rightarrow\infty}e^{-\frac{1}{p}\ln(p+1)}\\
    &=e^0\\
    &=1
\end{aligned}
\end{equation}
\begin{equation}
    \lim_{p\rightarrow\infty}(p+1)^{-1}=0
\end{equation}
Thus we have
\begin{equation}
    \lim_{p\rightarrow\infty}PoA(\mathcal{G}_p)
    =\lim_{p\rightarrow\infty}
    \frac{1}{1+(p+1)^{-\frac{1}{p}}\left[(p+1)^{-1}-1\right]}\\
    =\infty
\end{equation}
